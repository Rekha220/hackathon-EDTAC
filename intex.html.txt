    <script>
        // Sample predefined question bank to simulate AI topic quiz generation
        const sampleDatabase = {
            "default": [
                {
                    id: 1,
                    question: "What is the primary role of Chlorophyll during Photosynthesis?",
                    options: [
                        "Absorbing light energy from the sun",
                        "Absorbing water from the soil",
                        "Releasing carbon dioxide into the air",
                        "Breaking down glucose for immediate energy"
                    ],
                    correct: 0,
                    explanation: "Chlorophyll is the green pigment in plants that absorbs light energy (mostly blue and red wavelengths) and converts it into chemical energy."
                },
                {
                    id: 2,
                    question: "Which organelle inside plant cells is responsible for carrying out photosynthesis?",
                    options: [
                        "Mitochondria",
                        "Chloroplast",
                        "Ribosome",
                        "Endoplasmic Reticulum"
                    ],
                    correct: 1,
                    explanation: "Chloroplasts contain chlorophyll and the necessary enzymatic machinery to convert solar energy into chemical energy."
                },
                {
                    id: 3,
                    question: "What are the two primary end-products of Photosynthesis?",
                    options: [
                        "Carbon Dioxide and Water",
                        "Glucose and Oxygen",
                        "Nitrogen and ATP",
                        "Glucose and Carbon Dioxide"
                    ],
                    correct: 1,
                    explanation: "During photosynthesis, plants take in carbon dioxide and water to produce Glucose (food) and Oxygen (released into the atmosphere)."
                }
            ]
        };

        let currentQuizData = [];
        let userAnswers = {};

        function switchView(viewName) {
            document.getElementById('view-create').classList.add('hidden');
            document.getElementById('view-quiz').classList.add('hidden');
            document.getElementById('view-results').classList.add('hidden');

            if (viewName === 'create') {
                document.getElementById('view-create').classList.remove('hidden');
            } else if (viewName === 'quiz') {
                document.getElementById('view-quiz').classList.remove('hidden');
            } else if (viewName === 'results') {
                document.getElementById('view-results').classList.remove('hidden');
            }
        }

        function handleGenerateQuiz(e) {
            e.preventDefault();
            const subject = document.getElementById('subject-input').value;
            const topic = document.getElementById('topic-input').value;

            // Set Meta Details
            document.getElementById('quiz-topic-title').innerText = `${topic} (${subject})`;
            document.getElementById('quiz-badge').innerText = subject;

            // Load questions (In a full app, this would call an API/AI endpoint based on topic)
            currentQuizData = sampleDatabase.default;
            userAnswers = {};

            renderQuizQuestions();
            switchView('quiz');
        }

        function renderQuizQuestions() {
            const container = document.getElementById('questions-container');
            container.innerHTML = '';

            currentQuizData.forEach((q, index) => {
                const card = document.createElement('div');
                card.className = "bg-white border border-slate-200 rounded-2xl p-6 shadow-sm space-y-4";
                
                let optionsHTML = '';
                q.options.forEach((opt, optIndex) => {
                    optionsHTML += `
                        <label class="flex items-center space-x-3 p-3.5 border border-slate-200 rounded-xl cursor-pointer hover:bg-slate-50 transition">
                            <input type="radio" name="question-${q.id}" value="${optIndex}" onchange="recordAnswer(${q.id}, ${optIndex})" class="w-4 h-4 text-brand-600 focus:ring-brand-500">
                            <span class="text-sm font-medium text-slate-700">${opt}</span>
                        </label>
                    `;
                });

                card.innerHTML = `
                    <div class="flex items-start justify-between">
                        <span class="text-xs font-bold text-brand-600 bg-brand-50 px-2.5 py-1 rounded-md">Question ${index + 1}</span>
                    </div>
                    <h3 class="font-semibold text-slate-800 text-base">${q.question}</h3>
                    <div class="space-y-2.5 pt-2">
                        ${optionsHTML}
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function recordAnswer(questionId, optionIndex) {
            userAnswers[questionId] = optionIndex;
        }

        function submitQuiz() {
            if (Object.keys(userAnswers).length < currentQuizData.length) {
                if (!confirm("You haven't answered all questions. Are you sure you want to submit?")) {
                    return;
                }
            }

            renderResults();
            switchView('results');
        }

        function renderResults() {
            let correctCount = 0;
            const analysisContainer = document.getElementById('analysis-container');
            analysisContainer.innerHTML = '';

            currentQuizData.forEach((q, index) => {
                const selected = userAnswers[q.id];
                const isCorrect = selected === q.correct;
                if (isCorrect) correctCount++;

                const resultCard = document.createElement('div');
                resultCard.className = `bg-white border rounded-2xl p-6 shadow-sm space-y-4 ${
                    isCorrect ? 'border-emerald-200 bg-emerald-50/20' : 'border-rose-200 bg-rose-50/20'
                }`;

                let optionsAnalysis = '';
                q.options.forEach((opt, optIndex) => {
                    let badge = '';
                    let borderClass = 'border-slate-200 bg-white';

                    if (optIndex === q.correct) {
                        borderClass = 'border-emerald-500 bg-emerald-50 text-emerald-900 font-medium';
                        badge = `<span class="ml-auto text-xs font-bold text-emerald-700 bg-emerald-100 px-2 py-0.5 rounded">Correct Answer</span>`;
                    } else if (optIndex === selected && !isCorrect) {
                        borderClass = 'border-rose-500 bg-rose-50 text-rose-900 font-medium';
                        badge = `<span class="ml-auto text-xs font-bold text-rose-700 bg-rose-100 px-2 py-0.5 rounded">Your Selection (Incorrect)</span>`;
                    }

                    optionsAnalysis += `
                        <div class="flex items-center space-x-3 p-3 rounded-xl border ${borderClass} text-sm">
                            <span>${opt}</span>
                            ${badge}
                        </div>
                    `;
                });

                resultCard.innerHTML = `
                    <div class="flex items-center justify-between">
                        <span class="text-xs font-bold ${isCorrect ? 'text-emerald-700 bg-emerald-100' : 'text-rose-700 bg-rose-100'} px-2.5 py-1 rounded-md">
                            Question ${index + 1} - ${isCorrect ? 'Correct' : 'Needs Review'}
                        </span>
                    </div>
                    <h4 class="font-semibold text-slate-800">${q.question}</h4>
                    <div class="space-y-2 pt-1">${optionsAnalysis}</div>
                    
                    <!-- Conceptual Feedback / Correction Box -->
                    <div class="mt-4 p-4 rounded-xl bg-white border border-slate-200 text-xs text-slate-600 leading-relaxed space-y-1">
                        <div class="font-bold text-slate-800 flex items-center space-x-1.5">
                            <i class="fa-solid fa-lightbulb text-amber-500"></i>
                            <span>Concept Explanation:</span>
                        </div>
                        <p>${q.explanation}</p>
                    </div>
                `;

                analysisContainer.appendChild(resultCard);
            });

            // Update Header Score
            document.getElementById('score-circle').innerText = `${correctCount}/${currentQuizData.length}`;
            if (correctCount === currentQuizData.length) {
                document.getElementById('result-headline').innerText = "Perfect Mastery!";
                document.getElementById('result-subtext').innerText = "Excellent work! You have demonstrated strong understanding of this topic.";
            } else {
                document.getElementById('result-headline').innerText = "Review & Correct Your Mistakes";
                document.getElementById('result-subtext').innerText = "Carefully review the explanations below for questions you got wrong to fix your weak areas.";
            }
        }

        function retakeQuiz() {
            userAnswers = {};
            renderQuizQuestions();
            switchView('quiz');
        }
    </script>

    <script>
        // Sample predefined question bank to simulate AI topic quiz generation
        const sampleDatabase = {
            "default": [
                {
                    id: 1,
                    question: "What is the primary role of Chlorophyll during Photosynthesis?",
                    options: [
                        "Absorbing light energy from the sun",
                        "Absorbing water from the soil",
                        "Releasing carbon dioxide into the air",
                        "Breaking down glucose for immediate energy"
                    ],
                    correct: 0,
                    explanation: "Chlorophyll is the green pigment in plants that absorbs light energy (mostly blue and red wavelengths) and converts it into chemical energy."
                },
                {
                    id: 2,
                    question: "Which organelle inside plant cells is responsible for carrying out photosynthesis?",
                    options: [
                        "Mitochondria",
                        "Chloroplast",
                        "Ribosome",
                        "Endoplasmic Reticulum"
                    ],
                    correct: 1,
                    explanation: "Chloroplasts contain chlorophyll and the necessary enzymatic machinery to convert solar energy into chemical energy."
                },
                {
                    id: 3,
                    question: "What are the two primary end-products of Photosynthesis?",
                    options: [
                        "Carbon Dioxide and Water",
                        "Glucose and Oxygen",
                        "Nitrogen and ATP",
                        "Glucose and Carbon Dioxide"
                    ],
                    correct: 1,
                    explanation: "During photosynthesis, plants take in carbon dioxide and water to produce Glucose (food) and Oxygen (released into the atmosphere)."
                }
            ]
        };

        let currentQuizData = [];
        let userAnswers = {};

        function switchView(viewName) {
            document.getElementById('view-create').classList.add('hidden');
            document.getElementById('view-quiz').classList.add('hidden');
            document.getElementById('view-results').classList.add('hidden');

            if (viewName === 'create') {
                document.getElementById('view-create').classList.remove('hidden');
            } else if (viewName === 'quiz') {
                document.getElementById('view-quiz').classList.remove('hidden');
            } else if (viewName === 'results') {
                document.getElementById('view-results').classList.remove('hidden');
            }
        }

        function handleGenerateQuiz(e) {
            e.preventDefault();
            const subject = document.getElementById('subject-input').value;
            const topic = document.getElementById('topic-input').value;

            // Set Meta Details
            document.getElementById('quiz-topic-title').innerText = `${topic} (${subject})`;
            document.getElementById('quiz-badge').innerText = subject;

            // Load questions (In a full app, this would call an API/AI endpoint based on topic)
            currentQuizData = sampleDatabase.default;
            userAnswers = {};

            renderQuizQuestions();
            switchView('quiz');
        }

        function renderQuizQuestions() {
            const container = document.getElementById('questions-container');
            container.innerHTML = '';

            currentQuizData.forEach((q, index) => {
                const card = document.createElement('div');
                card.className = "bg-white border border-slate-200 rounded-2xl p-6 shadow-sm space-y-4";
                
                let optionsHTML = '';
                q.options.forEach((opt, optIndex) => {
                    optionsHTML += `
                        <label class="flex items-center space-x-3 p-3.5 border border-slate-200 rounded-xl cursor-pointer hover:bg-slate-50 transition">
                            <input type="radio" name="question-${q.id}" value="${optIndex}" onchange="recordAnswer(${q.id}, ${optIndex})" class="w-4 h-4 text-brand-600 focus:ring-brand-500">
                            <span class="text-sm font-medium text-slate-700">${opt}</span>
                        </label>
                    `;
                });

                card.innerHTML = `
                    <div class="flex items-start justify-between">
                        <span class="text-xs font-bold text-brand-600 bg-brand-50 px-2.5 py-1 rounded-md">Question ${index + 1}</span>
                    </div>
                    <h3 class="font-semibold text-slate-800 text-base">${q.question}</h3>
                    <div class="space-y-2.5 pt-2">
                        ${optionsHTML}
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function recordAnswer(questionId, optionIndex) {
            userAnswers[questionId] = optionIndex;
        }

        function submitQuiz() {
            if (Object.keys(userAnswers).length < currentQuizData.length) {
                if (!confirm("You haven't answered all questions. Are you sure you want to submit?")) {
                    return;
                }
            }

            renderResults();
            switchView('results');
        }

        function renderResults() {
            let correctCount = 0;
            const analysisContainer = document.getElementById('analysis-container');
            analysisContainer.innerHTML = '';

            currentQuizData.forEach((q, index) => {
                const selected = userAnswers[q.id];
                const isCorrect = selected === q.correct;
                if (isCorrect) correctCount++;

                const resultCard = document.createElement('div');
                resultCard.className = `bg-white border rounded-2xl p-6 shadow-sm space-y-4 ${
                    isCorrect ? 'border-emerald-200 bg-emerald-50/20' : 'border-rose-200 bg-rose-50/20'
                }`;

                let optionsAnalysis = '';
                q.options.forEach((opt, optIndex) => {
                    let badge = '';
                    let borderClass = 'border-slate-200 bg-white';

                    if (optIndex === q.correct) {
                        borderClass = 'border-emerald-500 bg-emerald-50 text-emerald-900 font-medium';
                        badge = `<span class="ml-auto text-xs font-bold text-emerald-700 bg-emerald-100 px-2 py-0.5 rounded">Correct Answer</span>`;
                    } else if (optIndex === selected && !isCorrect) {
                        borderClass = 'border-rose-500 bg-rose-50 text-rose-900 font-medium';
                        badge = `<span class="ml-auto text-xs font-bold text-rose-700 bg-rose-100 px-2 py-0.5 rounded">Your Selection (Incorrect)</span>`;
                    }

                    optionsAnalysis += `
                        <div class="flex items-center space-x-3 p-3 rounded-xl border ${borderClass} text-sm">
                            <span>${opt}</span>
                            ${badge}
                        </div>
                    `;
                });

                resultCard.innerHTML = `
                    <div class="flex items-center justify-between">
                        <span class="text-xs font-bold ${isCorrect ? 'text-emerald-700 bg-emerald-100' : 'text-rose-700 bg-rose-100'} px-2.5 py-1 rounded-md">
                            Question ${index + 1} - ${isCorrect ? 'Correct' : 'Needs Review'}
                        </span>
                    </div>
                    <h4 class="font-semibold text-slate-800">${q.question}</h4>
                    <div class="space-y-2 pt-1">${optionsAnalysis}</div>
                    
                    <!-- Conceptual Feedback / Correction Box -->
                    <div class="mt-4 p-4 rounded-xl bg-white border border-slate-200 text-xs text-slate-600 leading-relaxed space-y-1">
                        <div class="font-bold text-slate-800 flex items-center space-x-1.5">
                            <i class="fa-solid fa-lightbulb text-amber-500"></i>
                            <span>Concept Explanation:</span>
                        </div>
                        <p>${q.explanation}</p>
                    </div>
                `;

                analysisContainer.appendChild(resultCard);
            });

            // Update Header Score
            document.getElementById('score-circle').innerText = `${correctCount}/${currentQuizData.length}`;
            if (correctCount === currentQuizData.length) {
                document.getElementById('result-headline').innerText = "Perfect Mastery!";
                document.getElementById('result-subtext').innerText = "Excellent work! You have demonstrated strong understanding of this topic.";
            } else {
                document.getElementById('result-headline').innerText = "Review & Correct Your Mistakes";
                document.getElementById('result-subtext').innerText = "Carefully review the explanations below for questions you got wrong to fix your weak areas.";
            }
        }

        function retakeQuiz() {
            userAnswers = {};
            renderQuizQuestions();
            switchView('quiz');
        }
    </script>
